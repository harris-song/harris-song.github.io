<div id="not-found-msg">
Sorry, but the page you were trying to view does not exist.
</div>

<div id="r-viewer">
  <p>Loading document…</p>
  <noscript>
    JavaScript is required to view the embedded document. Please open the file directly.
  </noscript>
  <div id="r-viewer-container"></div>
</div>

<script>
  document.addEventListener('DOMContentLoaded', function () {
    try {
      console.log('[r-viewer] DOMContentLoaded fired');
      var path = location.pathname || '';
      if (path.indexOf('/r/') !== 0 || path.length <= 3) return;

      var rest = path.slice(3); // after '/r/'
      var target = '/files/' + rest;
      // Default to .pdf if no extension
      if (!/\.[a-z0-9]+$/i.test(target)) target += '.pdf';
      console.log('[r-viewer] computed target', { rest: rest, target: target });
      // Keep original target (no fragment modifications)

      // Update title for clarity
      try {
        var base = rest.replace(/\/+$/,'').split('/').pop();
        if (base && !/\.[a-z0-9]+$/i.test(base)) base += '.pdf';
        if (base) document.title = base + ' - ' + document.title;
      } catch (_) {}

      var container = document.getElementById('r-viewer-container');
      var wrapper = document.getElementById('r-viewer');
      if (!container || !wrapper) return;

      // Build <object>
      var objectEl = document.createElement('object');
      objectEl.setAttribute('data', target);
      objectEl.setAttribute('type', 'application/pdf');
      objectEl.setAttribute('width', '100%');
      objectEl.style.display = 'block';
      // Height will be set dynamically to keep 8.5x11 aspect ratio

      // Fallback <iframe>
      var iframeEl = document.createElement('iframe');
      iframeEl.setAttribute('src', target);
      iframeEl.setAttribute('width', '100%');
      iframeEl.style.display = 'block';
      // Height will be set dynamically to keep 8.5x11 aspect ratio
      objectEl.appendChild(iframeEl);

      // Fallback message
      var pMsg = document.createElement('p');
      var aOpen = document.createElement('a');
      aOpen.href = target;
      aOpen.target = '_blank';
      aOpen.rel = 'noopener';
      aOpen.textContent = 'Open the file';
      pMsg.appendChild(document.createTextNode('Cannot embed PDF. '));
      pMsg.appendChild(aOpen);
      pMsg.appendChild(document.createTextNode('.'));
      objectEl.appendChild(pMsg);

      // Append object
      container.innerHTML = '';
      container.appendChild(objectEl);

      // Link below
      // Prepare container for absolute controls
      container.style.position = 'relative';

      // Maintain 8.5x11 aspect ratio with a viewport cap
      var ASPECT = 11 / 8.5; // height / width
      function sizeEmbed() {
        try {
          var width = container.clientWidth || objectEl.clientWidth || 1600;
          var desired = Math.round(width * ASPECT);
          // subtract masthead height if present
          var masthead = document.querySelector('.masthead');
          var mastheadH = masthead ? masthead.offsetHeight : 0;
          var maxH = Math.floor(window.innerHeight - mastheadH - 16);
          var height = Math.min(desired, maxH);
          objectEl.style.height = height + 'px';
          iframeEl.style.height = height + 'px';
          container.style.height = height + 'px';
          container.style.overflow = 'hidden';
          var wrapper = document.getElementById('r-viewer');
          if (wrapper) {
            wrapper.style.minHeight = height + 'px';
            wrapper.style.height = height + 'px';
          }
        } catch (_) {}
      }
      sizeEmbed();
      window.addEventListener('resize', sizeEmbed);

      // No scroll overlay (restored previous behavior)

      // Hide default 404 message and title
      var notFound = document.getElementById('not-found-msg');
      if (notFound) notFound.style.display = 'none';
      var titleEl = document.querySelector('.page__title') || document.querySelector('h1');
      if (titleEl) titleEl.style.display = 'none';

      // Remove loading text
      var loadingEl = wrapper.querySelector('p');
      if (loadingEl) loadingEl.style.display = 'none';
    } catch (e) { /* no-op */ }
  });
  </script>
