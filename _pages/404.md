<div id="not-found-msg">
Sorry, but the page you were trying to view does not exist.
</div>

<div id="r-viewer">
  <p>Loading document…</p>
  <noscript>
    JavaScript is required to view the embedded document. Please open the file directly.
  </noscript>
  <div id="r-viewer-container"></div>
</div>

<script>
document.addEventListener('DOMContentLoaded', async function () {
  try {
    const path = location.pathname;
    if (!path.startsWith('/r/') || path.length <= 3) return;

    const rest = path.slice(3);
    let target = '/files/' + rest;
    if (!/\.[a-z0-9]+$/i.test(target)) target += '.pdf';
    target = new URL(target, location.origin).href; // ✅ absolute URL

    console.log('[r-viewer] trying to load', target);

    const container = document.getElementById('r-viewer-container');
    const wrapper = document.getElementById('r-viewer');
    if (!container || !wrapper) return;

    // Check that file exists before embedding
    const resp = await fetch(target, { method: 'HEAD' });
    if (!resp.ok) {
      container.innerHTML = '<p>Document not found.</p>';
      return;
    }

    // Build <object> element
    const objectEl = document.createElement('object');
    objectEl.data = target;
    objectEl.type = 'application/pdf';
    objectEl.width = '100%';
    objectEl.style.display = 'block';

    // Fallback link
    const fallback = document.createElement('p');
    fallback.innerHTML = `Cannot embed PDF. <a href="${target}" target="_blank" rel="noopener">Open the file</a>.`;
    objectEl.appendChild(fallback);

    container.innerHTML = '';
    container.appendChild(objectEl);

    // Resize logic
    const ASPECT = 11 / 8.5;
    const resize = () => {
      const width = container.clientWidth || 800;
      const height = Math.min(window.innerHeight - 16, width * ASPECT);
      objectEl.style.height = height + 'px';
      container.style.height = height + 'px';
    };
    resize();
    window.addEventListener('resize', resize);

    // Hide 404 defaults
    document.getElementById('not-found-msg')?.style.setProperty('display', 'none');
    document.querySelector('.page__title, h1')?.style.setProperty('display', 'none');
    wrapper.querySelector('p')?.style.setProperty('display', 'none');

  } catch (e) {
    console.error('[r-viewer] failed', e);
  }
});
</script>
